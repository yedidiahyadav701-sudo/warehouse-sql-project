CREATE VIEW vw_stock_details AS
SELECT p.product_name, w.warehouse_name, p.stock_qty, p.reorder_level,
       p.stock_qty * p.price AS stock_value
FROM products p
JOIN warehouses w ON p.warehouse_id = w.warehouse_id;
 
CREATE VIEW vw_low_stock AS
SELECT p.product_name, w.warehouse_name, p.stock_qty, p.reorder_level
FROM products p
JOIN warehouses w ON p.warehouse_id = w.warehouse_id
WHERE p.stock_qty <= p.reorder_level;
 
CREATE VIEW vw_warehouse_summary AS
SELECT w.warehouse_name, w.location,
       COUNT(p.product_id)                     AS total_products,
       COALESCE(SUM(p.stock_qty), 0)           AS total_units,
       COALESCE(SUM(p.stock_qty * p.price), 0) AS stock_value
FROM warehouses w
LEFT JOIN products p ON w.warehouse_id = p.warehouse_id
GROUP BY w.warehouse_id, w.warehouse_name, w.location;
 
CREATE VIEW vw_outbound_details AS
SELECT o.outbound_id, c.customer_name, p.product_name, o.quantity,
       o.quantity * p.price AS total_amount, o.dispatched_date
FROM outbound o
JOIN customers c ON o.customer_id = c.customer_id
JOIN products  p ON o.product_id  = p.product_id;
